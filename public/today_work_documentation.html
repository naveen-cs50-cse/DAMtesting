lib/investigation/types.ts • Verdict Resolution Logic 
export type QueryVerdictStatus = 'right' | 'false' | 'unverified';

export function getQueryVerdictStatus(meta?: InvestigationMeta | null): QueryVerdictStatus | null {
  if (!meta || meta.taskType === 'conversational' || !meta.verdicts) return null;
  if (meta.verdicts.verdictStatus) return meta.verdicts.verdictStatus; // Evaluated by AI Challenger

  const { factualCorrectness, scamRisk } = meta.verdicts;
  if (factualCorrectness === 'Contradicted' || scamRisk === 'High') return 'false';
  if (factualCorrectness === 'Supported') return 'right';
  return 'unverified';
}
02

AI Adversarial Fact-Checking Challenger Engine

The Problem: When claims like "prime minister of india modi is dead" or "2000 are banned in india" were investigated, keyword search failed to return exact matching snippets. The heuristic engine treated missing evidence as "Insufficient", defaulting to a white dot ("Needs Verification") rather than catching the obvious falsehood.

The Solution: Implemented lib/investigation/aiChallenger.ts which calls OpenAI (gpt-4o-mini) in Stage 4 of the investigation pipeline. It subjects every claim to an adversarial fact-check against established historical facts, Indian statutes (e.g. RBI Clean Note Policy, Section 56 of Electricity Act), and known cyber scams.

05

Stability & Architectural Fixes

  • 1. Next.js SSR Hydration Error Resolution (components/Topbar.tsx):
    Root Cause: Next.js server rendered 2 header elements, but on the client an unmounted localStorage hook immediately rendered an evidence counter badge, causing React hydration to throw error #418.
    Fix: Wrapped the client evidence badge in a mounted state hook: const [mounted, setMounted] = useState(false); useEffect(() => setMounted(true), []);
  • 2. API Key Shadowing & Resilient Disk Fallback (app/api/investigate/stream/route.ts):
    Root Cause: Windows user environment variables (OPENAI_API_KEY=fe_oa_...) shadowed project .env keys, causing 401 Unauthorized errors from OpenAI.
    Fix: Implemented getEffectiveOpenAIKey() and getEffectiveTavilyKey() which directly inspect and parse .env from disk whenever environment variables do not match expected prefixes.
  • 3. Stream Route History Scope Preservation:
    Fix: Restored the history payload variable in app/api/investigate/stream/route.ts, preventing undefined runtime evaluation during conversational follow-ups.

import { InvestigationEvidence, InvestigationVerdicts, QueryVerdictStatus } from './types';

export interface AIChallengerResult {
  verdictStatus: QueryVerdictStatus;
  primaryRationale: string;
  scamThreatIdentified: boolean;
  debunkNotice?: string;
  statutoryReference?: string;
}

export async function evaluateClaimWithAI(
  claim: string,
  apiKey?: string,
  evidenceSnippets?: InvestigationEvidence[]
): Promise<AIChallengerResult | null> {
  if (!apiKey) return null;
  // Evaluates veracity against established world knowledge, Indian gazettes, and cyber threat indicators
  const response = await fetch('https://api.openai.com/v1/chat/completions', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`
    },
    body: JSON.stringify({
      model: 'gpt-4o-mini',
      response_format: { type: 'json_object' },
      messages: [
        { role: 'system', content: 'You are an adversarial regulatory fact-checker and cyber scam investigator.' },
        { role: 'user', content: claim }
      ],
      temperature: 0.1
    })
  });
  // Returns definitive verdictStatus: 'right' | 'false' | 'unverified'
}
